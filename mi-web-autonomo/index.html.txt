  <script>
    // Datos de productos
    const products = [
      { id: 1, title: 'Kit UI Figma Dashboard', category: 'plantillas', price: 29.00, tag: 'Popular' },
      { id: 2, title: 'Guía de Headless CMS para Freelancers', category: 'guías', price: 15.00, tag: 'PDF' },
      { id: 3, title: 'Plantilla Portfolio Minimalista', category: 'plantillas', price: 39.00, tag: 'Vercel Ready' },
      { id: 4, title: 'Manual de Identidad Visual (Plantilla)', category: 'plantillas', price: 19.00, tag: 'Figma' },
      { id: 5, title: 'Checklist de Seguridad Web 2026', category: 'guías', price: 9.00, tag: 'Ebook' }
    ];

    let cart = [];

    // Renderizado de productos
    function renderProducts(items) {
      const grid = document.getElementById('product-grid');
      grid.innerHTML = items.map(p => `
        <div class="bg-white border border-slate-200 rounded-2xl p-5 hover:shadow-lg transition-all flex flex-col justify-between">
          <div>
            <div class="flex justify-between items-start">
              <span class="text-[11px] font-semibold tracking-wider text-indigo-600 bg-indigo-50 px-2 py-0.5 rounded-md uppercase">${p.tag}</span>
              <span class="font-bold text-slate-900">${p.price.toFixed(2)} €</span>
            </div>
            <h4 class="font-bold text-slate-800 mt-4 text-base">${p.title}</h4>
          </div>
          <button onclick="addToCart(${p.id})" class="mt-6 w-full py-2 bg-slate-900 hover:bg-slate-800 text-white rounded-lg text-xs font-semibold flex items-center justify-center gap-2 transition-colors">
            <i data-lucide="plus" class="w-4 h-4"></i> Añadir a la cesta
          </button>
        </div>
      `).join('');
      lucide.createIcons();
    }

    // Filtros
    function filterProducts(category) {
      document.querySelectorAll('.filter-btn').forEach(btn => {
        if (btn.getAttribute('data-category') === category) {
          btn.className = 'filter-btn px-4 py-1.5 text-xs font-semibold rounded-full bg-slate-900 text-white';
        } else {
          btn.className = 'filter-btn px-4 py-1.5 text-xs font-semibold rounded-full bg-slate-200 text-slate-700 hover:bg-slate-300';
        }
      });

      if (category === 'todos') {
        renderProducts(products);
      } else {
        renderProducts(products.filter(p => p.category === category));
      }
    }

    // Carrito
    function addToCart(id) {
      const item = products.find(p => p.id === id);
      cart.push(item);
      updateCartUI();
    }

    function updateCartUI() {
      document.getElementById('cart-count').innerText = cart.length;
      const list = document.getElementById('cart-items');
      
      if (cart.length === 0) {
        list.innerHTML = '<p class="text-sm text-slate-400 text-center py-8">La cesta está vacía</p>';
      } else {
        list.innerHTML = cart.map((item, index) => `
          <div class="flex justify-between items-center text-sm">
            <div>
              <p class="font-medium text-slate-800">${item.title}</p>
              <p class="text-slate-500 text-xs">${item.price.toFixed(2)} €</p>
            </div>
            <button onclick="removeFromCart(${index})" class="text-rose-500 hover:text-rose-700 text-xs">Eliminar</button>
          </div>
        `).join('');
      }

      const total = cart.reduce((acc, curr) => acc + curr.price, 0);
      document.getElementById('cart-total').innerText = total.toFixed(2) + ' €';
    }

    function removeFromCart(index) {
      cart.splice(index, 1);
      updateCartUI();
    }

    function toggleCart() {
      const modal = document.getElementById('cart-modal');
      modal.classList.toggle('hidden');
    }

    function checkout() {
      if (cart.length === 0) return alert('La cesta está vacía.');
      alert('¡Redirigiendo a pasarela segura (simulada)! Total: ' + document.getElementById('cart-total').innerText);
    }

    document.getElementById('cart-btn').addEventListener('click', toggleCart);

    // Inicialización
    renderProducts(products);
    lucide.createIcons();
  </script>
